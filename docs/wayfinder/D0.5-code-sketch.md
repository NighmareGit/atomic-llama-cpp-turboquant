static bool llama_gpipe_enabled() {
    static int v = -1;
    if (v < 0) {
        const char * e = getenv("GGML_SCHED_GPIPE");
        v = (e != nullptr && atoi(e) != 0) ? 1 : 0;
    }
    return v != 0;
}

## 7. `ggml/src/ggml-backend.cpp` -- `ggml_sched_gpipe_init()`

New function near `ggml_backend_sched_new()` (~line 2830):

```cpp
// Initialize GPipe-specific event arrays.
// Creates n_stages events for producer-consumer signaling across tokens.
void ggml_sched_gpipe_init(ggml_backend_sched_t sched, int n_stages) {
    GGML_ASSERT(sched);
    GGML_ASSERT(n_stages > 0 && n_stages <= GGML_SCHED_MAX_STAGES);

    // GPipe events are indexed by stage, not backend/copy.
    // They are created on the last backend (gather device) by default.
    // The gather backend is the last one in sched->backends[].
    const int gather_bid = sched->n_backends - 1;
    ggml_backend_t gather_backend = sched->backends[gather_bid];

    for (int s = 0; s < n_stages; s++) {
        sched->gpipe_events[s] = ggml_backend_event_new(gather_backend->device);
    }
}
```

## 8. `src/llama-context.cpp` -- `llama_decode_gpipe_impl()`

New private implementation function, called from `llama_context::decode()`:

```cpp
int llama_context::decode_gpipe_impl(const llama_batch & batch_inp,
                                      const float * h_prev) {
    GGML_ASSERT(gpipe.enabled);
    GGML_ASSERT(gpipe.n_stages >= 2);

    // Set MTP hidden state if provided
    gpipe_h_prev = h_prev;

    // Stage 0: embed + RPC compute
    // This reuses process_ubatch() for the actual graph compute,
    // but replaces the per-split serial dispatch with stage-aware dispatch.
    //
    // The key difference from classic mode:
    //   1. Stage 0 runs compute_splits for splits 0..n_splits-2 (embed+RPC)
    //   2. After stage 0: event_record(compute_done)
    //   3. Stage 1 runs compute_splits for split n_splits-1 (gather)
    //   4. After stage 1: event_record(kv_ready)
    //   5. Between stages: check if T+1's embed can be dispatched early
    //
    // For initial implementation:
    //   - Run full compute_splits as today (no sub-split gating)
    //   - After compute_splits, record stage events
    //   - Advance cur_stage
    //   - Return early if we're in compute stage and gather can overlap

    // ... (implementation details in D2.x)

    return 0;
}
```
